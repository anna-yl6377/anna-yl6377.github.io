1 Bar Plot: Average Inspection Score by Borough

plot_ly(
  borough_score,
  x = ~borough,
  y = ~average_score,
  type = "bar",
  color = ~borough,
  text = ~paste(
    "Average score:", round(average_score, 2),
    "<br>Median score:", median_score,
    "<br>Records:", n
  ),
  hoverinfo = "text"
) |>
  layout(
    title = "Average Inspection Score by Borough",
    xaxis = list(title = "Borough"),
    yaxis = list(title = "Average Inspection Score"),
    showlegend = FALSE
  )
